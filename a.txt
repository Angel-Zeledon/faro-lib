  BUG 4 — isAuthenticated() no detecta tokens expirados que están cerca del límite
  Archivo: C:\Users\Jahir\Documents\forecasting\Frontend\src\lib\auth.ts
  Función: isAuthenticated

  Fix: añadir margen de 30 segundos y validar que el payload tiene los campos mínimos:
    export function isAuthenticated(): boolean {
      const token = getToken()
      if (!token) return false
      try {
        const payload = JSON.parse(atob(token.split('.')[1]))
        // Validar campos mínimos esperados
        if (!payload.sub || !payload.tenant_id || !payload.exp) return false
        // Margen de 30s para evitar que expire durante requests en vuelo
        return payload.exp > (Date.now() / 1000) + 30
      } catch {

      }
    }
